cmake_minimum_required(VERSION 3.22)
project(Tasked VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt6 6.8 REQUIRED COMPONENTS Quick)
qt_standard_project_setup(REQUIRES 6.8)
qt_add_executable(Tasked MANUAL_FINALIZATION src/main.cpp)
target_sources(Tasked PRIVATE src/app_launcher.cpp src/app_launcher.h src/shell_icon_provider.cpp src/shell_icon_provider.h src/platform/dock_platform.h)
if(WIN32)
    target_sources(Tasked PRIVATE src/platform/dock_platform_win.cpp)
    target_link_libraries(Tasked PRIVATE shell32 user32)
else()
    target_sources(Tasked PRIVATE src/platform/dock_platform_stub.cpp)
endif()
qt_add_qml_module(Tasked URI Tasked VERSION 1.0 QML_FILES qml/Main.qml)
set_target_properties(Tasked PROPERTIES WIN32_EXECUTABLE TRUE)
target_link_libraries(Tasked PRIVATE Qt6::Quick)
qt_import_qml_plugins(Tasked)
qt_finalize_executable(Tasked)
