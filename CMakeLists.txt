cmake_minimum_required(VERSION 3.22)
project(Tasked VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin")
find_package(Qt6 6.8 REQUIRED COMPONENTS Quick)
qt_standard_project_setup(REQUIRES 6.8)
qt_add_executable(Tasked MANUAL_FINALIZATION src/main.cpp)
target_sources(Tasked PRIVATE src/app_launcher.cpp src/app_launcher.h src/app_settings.cpp src/app_settings.h src/extensions_model.cpp src/extensions_model.h src/preview_controller.cpp src/preview_controller.h src/shell_icon_provider.cpp src/shell_icon_provider.h src/platform/dock_platform.h src/platform/windows_shell.cpp src/platform/windows_shell.h src/platform/tray_icon_provider.cpp src/platform/tray_icon_provider.h)
qt_add_resources(Tasked resources/Tasked.qrc)
if(WIN32)
    target_sources(Tasked PRIVATE src/platform/dock_platform_win.cpp src/platform/tray_icon_hook_client.cpp src/platform/tray_icon_hook_client.h src/platform/tray_icon_hook_protocol.h src/platform/taskbar_symbol_resolver.cpp src/platform/taskbar_symbol_resolver.h resources/Tasked.rc resources/resource.h)
    target_link_libraries(Tasked PRIVATE shell32 user32 dwmapi uiautomationcore ole32 gdi32 dbghelp winhttp)
    add_library(TaskedTrayHook SHARED src/platform/tray_icon_hook_dll.cpp src/platform/tray_icon_hook_protocol.h)
    target_include_directories(TaskedTrayHook PRIVATE src/platform)
    target_link_libraries(TaskedTrayHook PRIVATE shell32 user32 ole32 windowsapp)
    set_target_properties(TaskedTrayHook PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin")
    add_dependencies(Tasked TaskedTrayHook)
else()
    target_sources(Tasked PRIVATE src/platform/dock_platform_stub.cpp)
endif()
qt_add_qml_module(Tasked URI Tasked VERSION 1.0 QML_FILES qml/Main.qml)
set_target_properties(Tasked PROPERTIES WIN32_EXECUTABLE TRUE RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin")
target_link_libraries(Tasked PRIVATE Qt6::Quick)
qt_import_qml_plugins(Tasked)
qt_finalize_executable(Tasked)
